-- 日记表
CREATE TABLE diary (
  id bigint generated always as identity primary key,
  created_at timestamptz default now(),
  date text not null,
  raw_input text,
  diary_text text not null
);

alter table diary enable row level security;
create policy "allow insert" on diary for insert with check (true);
create policy "allow select" on diary for select using (true);

-- 聊天记录同步表
CREATE TABLE chat_history (
  id bigint generated always as identity primary key,
  created_at timestamptz default now(),
  updated_at timestamptz default now(),
  sync_code text unique not null,
  messages jsonb not null,
  device_name text
);

alter table chat_history enable row level security;
create policy "allow all" on chat_history for all using (true) with check (true);
