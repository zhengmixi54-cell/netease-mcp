-- 日记表
CREATE TABLE diary (
  id bigint generated always as identity primary key,
  created_at timestamptz default now(),
  date text not null,
  raw_input text,
  diary_text text not null
);

alter table diary enable row level security;
create policy "allow insert" on diary for insert with check (true);
create policy "allow select" on diary for select using (true);

-- 聊天记录同步表
CREATE TABLE chat_history (
  id bigint generated always as identity primary key,
  created_at timestamptz default now(),
  updated_at timestamptz default now(),
  sync_code text unique not null,
  messages jsonb not null,
  device_name text
);

alter table chat_history enable row level security;
create policy "allow all" on chat_history for all using (true) with check (true);

-- 心情打卡表
CREATE TABLE mood_entries (
  id bigint generated always as identity primary key,
  created_at timestamptz default now(),
  sync_code text not null,
  date text not null,
  mood text not null,
  note text,
  unique(sync_code, date)
);

alter table mood_entries enable row level security;
create policy "allow all" on mood_entries for all using (true) with check (true);

-- 记忆库表
CREATE TABLE memory_bank (
  id bigint generated always as identity primary key,
  created_at timestamptz default now(),
  updated_at timestamptz default now(),
  sync_code text unique not null,
  memories jsonb not null
);

alter table memory_bank enable row level security;
create policy "allow all" on memory_bank for all using (true) with check (true);
